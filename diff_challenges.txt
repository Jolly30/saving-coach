+    fun autoSkipMissedDays(challengeId: String) {
+        val challenge = uiState.value.challengesList.find { it.id == challengeId } ?: return
+        if (challenge.isCompleted) return
+
+        val parts = challenge.lastDepositDate.split("|")
+        val lastDateStr = parts.getOrNull(0) ?: ""
+        val completedSteps = parts.getOrNull(1)?.toIntOrNull() ?: 0
+        val duration = parts.getOrNull(2)?.toIntOrNull() ?: 30
+
+        if (lastDateStr.isEmpty()) return // No deposits yet, or start day
+
+        try {
+            val lastDate = LocalDate.parse(lastDateStr)
+            val today = LocalDate.now()
+            val daysDiff = java.time.temporal.ChronoUnit.DAYS.between(lastDate, today).toInt()
+
+            if (daysDiff > 1 && completedSteps < duration) {
+                val daysToSkip = minOf(daysDiff - 1, duration - completedSteps)
+                if (daysToSkip > 0) {
+                    viewModelScope.launch {
+                        var newCompletedSteps = completedSteps
+                        for (i in 1..daysToSkip) {
+                            newCompletedSteps++
+                            val skipDate = lastDate.plusDays(i.toLong()).toString()
+                            
+                            val deposit = SavingsDeposit(
+                                id = java.util.UUID.randomUUID().toString(),
+                                challengeId = challengeId,
+                                amount = 0.0,
+                                date = skipDate,
+                                note = "Skipped",
+                                createdAt = System.currentTimeMillis() - ((daysToSkip - i) * 1000) // Ensure stable ordering
+                            )
+                            repository.addDeposit(userId, challengeId, deposit)
+                        }
+
+                        // Update challenge
+                        val updatedChallenge = challenge.copy(
+                            lastDepositDate = lastDate.plusDays(daysToSkip.toLong()).toString() + "|" + newCompletedSteps + "|" + duration,
+                            isCompleted = newCompletedSteps >= duration,
+                            isActive = newCompletedSteps < duration
+                        )
+                        repository.createChallenge(updatedChallenge)
+                    }
+                }
+            }
+        } catch (e: Exception) {
+            // Ignore parse errors
+        }
